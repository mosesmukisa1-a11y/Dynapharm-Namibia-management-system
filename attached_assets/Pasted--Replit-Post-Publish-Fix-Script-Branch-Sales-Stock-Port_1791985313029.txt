```tsx
function canReviewSale(state: SaleState) {
  if (!state.customerType) return { ok: false, message: "Select customer type." };
  if (!state.cart.length) return { ok: false, message: "Add at least one product." };
  if (state.customerType === "walk_in" && !state.customerName?.trim()) {
    return { ok: false, message: "Enter customer name." };
  }
  if (!state.paymentMethod) return { ok: false, message: "Select payment method." };
  return { ok: true };
}

function handleReviewSale() {
  const validation = canReviewSale(saleState);
  if (!validation.ok) {
    setFormError(validation.message);
    return;
  }
  setFormError(null);
  setSaleStep("review");
}
```

- Shell renders:

Required fixes:

- Every stock view must render its content after:
  - top-nav click into Stock
  - direct `/stock?view=...` URL
  - browser reload
  - switching from Branch/Finance/Director/MIS to Stock
- If data fails to load, show a visible error panel with retry, not a blank content area.
- Do not hide the view body behind a stale `isLoading`, missing role, undefined location scope, or failed suspense state.

Add a reusable guard:

```tsx
function StockViewBoundary({
  view,
  loading,
  error,
  children,
}: {
  view: string;
  loading: boolean;
  error?: Error | string | null;
  children: React.ReactNode;
}) {
  if (loading) {
    return <div className="stock-panel">Loading {view}...</div>;
  }

  if (error) {
    return (
      <div className="stock-panel stock-panel-error">
        <h3>Could not load {view}</h3>
        <p>{String(error)}</p>
        <button onClick={() => window.location.reload()}>Retry</button>
      </div>
    );
  }

  return <>{children}</>;
}
```

Keep this, but improve cost evidence:

```ts
function classifyCostEvidence(line: {
  originalReceivedDPC?: number | null;
  originalReceivedCost?: number | null;
}) {
  const hasDpc = line.originalReceivedDPC !== null && line.originalReceivedDPC !== undefined && Number(line.originalReceivedDPC) > 0;
  const hasCost = line.originalReceivedCost !== null && line.originalReceivedCost !== undefined && Number(line.originalReceivedCost) > 0;

  if (hasDpc && hasCost) return "original";
  if (hasDpc || hasCost) return "partial";
  return "missing";
}
```
