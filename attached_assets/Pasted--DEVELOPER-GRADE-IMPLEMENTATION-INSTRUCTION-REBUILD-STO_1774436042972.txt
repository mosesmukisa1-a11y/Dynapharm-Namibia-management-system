```ts
export async function postStockMovement(input: StockMovementInput): Promise<void> {
  await validateMovement(input);

  await db.stock_movements.create({
    data: {
      ...input,
      status: "posted",
    },
  });

  if (input.sourceLocationType && input.sourceLocationId) {
    await recomputeInventoryBalance({
      locationType: input.sourceLocationType,
      locationId: input.sourceLocationId,
      productId: input.productId,
      batchNo: input.batchNo,
      expiryDate: input.expiryDate,
    });
  }

  if (input.destinationLocationType && input.destinationLocationId) {
    await recomputeInventoryBalance({
      locationType: input.destinationLocationType,
      locationId: input.destinationLocationId,
      productId: input.productId,
      batchNo: input.batchNo,
      expiryDate: input.expiryDate,
    });
  }

  if (input.sourceLocationType && input.sourceLocationId) {
    await recomputeLocationSummary({
      locationType: input.sourceLocationType as "warehouse" | "branch" | "field",
      locationId: input.sourceLocationId,
    }).catch(() => null);
  }

  if (input.destinationLocationType && input.destinationLocationId) {
    await recomputeLocationSummary({
      locationType: input.destinationLocationType as "warehouse" | "branch" | "field",
      locationId: input.destinationLocationId,
    }).catch(() => null);
  }
}
```

```ts
const INWARD_TYPES = new Set([
  "supplier_import_receipt",
  "warehouse_receipt",
  "branch_receipt",
  "transfer_in",
  "return_in",
  "positive_adjustment",
]);

const OUTWARD_TYPES = new Set([
  "warehouse_dispatch",
  "transfer_out",
  "walkin_sale",
  "walkin_member_sale",
  "walkin_non_member_sale",
  "online_order_fulfilment",
  "prescription_dispense",
  "write_off",
  "return_out",
  "negative_adjustment",
]);
```

```ts
export async function recomputeInventoryBalance(key: StockKey): Promise<void> {
  const movements = await db.stock_movements.findMany({
    where: {
      status: "posted",
      productId: key.productId,
      batchNo: key.batchNo,
      expiryDate: new Date(key.expiryDate),
      OR: [
        {
          sourceLocationType: key.locationType,
          sourceLocationId: key.locationId,
        },
        {
          destinationLocationType: key.locationType,
          destinationLocationId: key.locationId,
        },
      ],
    },
    orderBy: { movementDate: "asc" },
  });

  let inwardQty = 0;
  let outwardQty = 0;
  let productName = "";
  let productDescription = "";
  let unitDP = 0;
  let unitCIF = 0;
  let lastMovementDate: Date | null = null;

  for (const m of movements) {
    productName = m.productName;
    productDescription = m.productDescription;
    unitDP = m.unitDP ?? unitDP;
    unitCIF = m.unitCIF ?? unitCIF;
    lastMovementDate = m.movementDate;

    const isInboundToLocation =
      m.destinationLocationType === key.locationType &&
      m.destinationLocationId === key.locationId &&
      INWARD_TYPES.has(m.movementType);

    const isOutboundFromLocation =
      m.sourceLocationType === key.locationType &&
      m.sourceLocationId === key.locationId &&
      OUTWARD_TYPES.has(m.movementType);

    if (isInboundToLocation) inwardQty += m.quantity;
    if (isOutboundFromLocation) outwardQty += m.quantity;
  }

  const closingQty = inwardQty - outwardQty;

  await db.inventory_balances.upsert({
    where: {
      unique_inventory_key: {
        locationType: key.locationType,
        locationId: key.locationId,
        productId: key.productId,
        batchNo: key.batchNo,
        expiryDate: new Date(key.expiryDate),
      },
    },
    create: {
      locationType: key.locationType,
      locationId: key.locationId,
      productId: key.productId,
      productName,
      productDescription,
      batchNo: key.batchNo,
      expiryDate: new Date(key.expiryDate),
      openingQty: 0,
      inwardQty,
      outwardQty,
      closingQty,
      unitDP,
      unitCIF,
      stockValueDP: closingQty * unitDP,
      stockValueCIF: closingQty * unitCIF,
      lastMovementDate,
    },
    update: {
      productName,
      productDescription,
      inwardQty,
      outwardQty,
      closingQty,
      unitDP,
      unitCIF,
      stockValueDP: closingQty * unitDP,
      stockValueCIF: closingQty * unitCIF,
      lastMovementDate,
    },
  });
}
```

```ts
export async function getAvailableStock(key: StockKey): Promise<number> {
```

```ts
async function validateAvailableStockForOutward(input: StockMovementInput) {
  if (!input.sourceLocationType || !input.sourceLocationId) return;

  const outwardTypes = [
    "warehouse_dispatch",
    "transfer_out",
    "walkin_sale",
    "walkin_member_sale",
    "walkin_non_member_sale",
    "online_order_fulfilment",
    "prescription_dispense",
    "write_off",
    "return_out",
    "negative_adjustment",
  ];

  if (!outwardTypes.includes(input.movementType)) return;

  const available = await getAvailableStock({
    locationType: input.sourceLocationType,
    locationId: input.sourceLocationId,
    productId: input.productId,
    batchNo: input.batchNo,
    expiryDate: input.expiryDate,
  });

  if (available < input.quantity) {
    throw new Error(
      `Insufficient stock for ${input.productName}. Available: ${available}, Requested: ${input.quantity}`
    );
  }
}
```

# 8. REQUIRED VALIDATION RULES

```ts
async function validateMovement(input: StockMovementInput) {
  if (!input.productId) throw new Error("Missing productId");
  if (!input.batchNo) throw new Error("Batch number is required");
  if (!input.expiryDate) throw new Error("Expiry date is required");
  if (!input.quantity || input.quantity <= 0) throw new Error("Quantity must be greater than zero");
  if (!input.unitDP || input.unitDP < 0) throw new Error("DP unit price is required");

  await validateAvailableStockForOutward(input);
}
```

```ts
async function dispensePrescriptionItem(params: {
  prescriptionId: string;
  prescriptionItemId: string;
  branchId: string;
  batchNo: string;
  expiryDate: string;
  dispenseQty: number;
  unitDP: number;
  userId: string;
  receiptId: string;
}) {
  const item = await db.prescription_items.findUniqueOrThrow({
    where: { id: params.prescriptionItemId },
  });

  const newDispensedTotal = item.dispensedQtyTotal + params.dispenseQty;

  if (newDispensedTotal > item.prescribedQty) {
    throw new Error("Cannot dispense more than prescribed quantity");
  }

  const remainingQty = item.prescribedQty - newDispensedTotal;

  let dispensingStatus: "not_dispensed" | "partially_dispensed" | "fully_dispensed" = "not_dispensed";
  if (newDispensedTotal === 0) dispensingStatus = "not_dispensed";
  else if (remainingQty === 0) dispensingStatus = "fully_dispensed";
  else dispensingStatus = "partially_dispensed";

  await postStockMovement({
    movementDate: new Date(),
    movementType: "prescription_dispense",
    sourceLocationType: "branch",
    sourceLocationId: params.branchId,
    destinationLocationType: null,
    destinationLocationId: null,
    productId: item.productId,
    productName: item.productName,
    productDescription: item.productName,
    batchNo: params.batchNo,
    expiryDate: params.expiryDate,
    quantity: params.dispenseQty,
    unitDP: params.unitDP,
    referenceType: "prescription",
    referenceId: params.prescriptionId,
    createdBy: params.userId,
    notes: "Prescription dispensed",
  });

  await db.prescription_items.update({
    where: { id: item.id },
    data: {
      dispensedQtyTotal: newDispensedTotal,
      remainingQty,
      dispensingStatus,
    },
  });
}
```

```ts
async function getBranchStockCard(branchId: string, startDate: Date, endDate: Date) {
  const balances = await db.inventory_balances.findMany({
    where: {
      locationType: "branch",
      locationId: branchId,
      closingQty: { gt: 0 },
    },
  });

  const movements = await db.stock_movements.findMany({
    where: {
      status: "posted",
      OR: [
        { sourceLocationType: "branch", sourceLocationId: branchId },
        { destinationLocationType: "branch", destinationLocationId: branchId },
      ],
      movementDate: {
        gte: startDate,
        lte: endDate,
      },
    },
  });

  const productCount = balances.filter(b => b.closingQty > 0).length;
  const closingStock = balances.reduce((sum, b) => sum + b.closingQty, 0);
  const stockValueDP = balances.reduce((sum, b) => sum + b.stockValueDP, 0);

  const salesUnits = movements
    .filter(m =>
      ["walkin_sale", "walkin_member_sale", "walkin_non_member_sale", "online_order_fulfilment"].includes(m.movementType) &&
      m.sourceLocationType === "branch" &&
      m.sourceLocationId === branchId
    )
    .reduce((sum, m) => sum + m.quantity, 0);

  const dispensedUnits = movements
    .filter(m =>
      m.movementType === "prescription_dispense" &&
      m.sourceLocationType === "branch" &&
      m.sourceLocationId === branchId
    )
    .reduce((sum, m) => sum + m.quantity, 0);

  return {
    productCount,
    closingStock,
    stockValueDP,
    salesUnits,
    dispensedUnits,
  };
}
```

```ts
async function getOpeningStock(params: StockPeriodKey): Promise<number> {
  const allBefore = await db.stock_movements.findMany({
    where: {
      productId: params.productId,
      batchNo: params.batchNo,
      expiryDate: new Date(params.expiryDate),
      movementDate: { lt: new Date(params.startDate) },
      status: "posted",
      OR: [
        {
          sourceLocationType: params.locationType,
          sourceLocationId: params.locationId,
        },
        {
          destinationLocationType: params.locationType,
          destinationLocationId: params.locationId,
        },
      ],
    },
  });

  let inward = 0;
  let outward = 0;

  for (const m of allBefore) {
    const inbound =
      m.destinationLocationType === params.locationType &&
      m.destinationLocationId === params.locationId &&
      INWARD_TYPES.has(m.movementType);

    const outbound =
      m.sourceLocationType === params.locationType &&
      m.sourceLocationId === params.locationId &&
      OUTWARD_TYPES.has(m.movementType);

    if (inbound) inward += m.quantity;
    if (outbound) outward += m.quantity;
  }

  return inward - outward;
}
```
