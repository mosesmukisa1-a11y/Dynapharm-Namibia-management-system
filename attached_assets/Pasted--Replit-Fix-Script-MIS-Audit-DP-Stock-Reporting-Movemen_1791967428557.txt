## Implementation Pattern

Find the sale completion handler. It is likely the function behind `Complete Sale`.

Refactor it so a completed sale writes all related evidence in one atomic operation/transaction:

```ts
type SaleLineEvidence = {
  productId: string;
  productName: string;
  batchNo?: string | null;
  expiryDate?: string | null;
  quantity: number;
  dpUnitPrice: number;
  chargedUnitPrice: number;
  bv: number;
};

function toMoney(n: number) {
  return Math.round(Number(n || 0) * 100) / 100;
}

function namibiaBusinessDate(date = new Date()) {
  const parts = new Intl.DateTimeFormat("en-CA", {
    timeZone: "Africa/Windhoek",
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  }).formatToParts(date);
  const get = (type: string) => parts.find((p) => p.type === type)?.value;
  return `${get("year")}-${get("month")}-${get("day")}`;
}

function namibiaDisplayTimestamp(date = new Date()) {
  return new Intl.DateTimeFormat("en-GB", {
    timeZone: "Africa/Windhoek",
    year: "numeric",
    month: "short",
    day: "2-digit",
    hour: "2-digit",
    minute: "2-digit",
    hour12: false,
  }).format(date);
}
```

Inside `completeSale(...)`, calculate and persist both charged and DP evidence:

```ts
const now = new Date();
const businessDate = namibiaBusinessDate(now);

const saleLines = cartItems.map((item) => {
  const quantity = Number(item.quantity || 1);
  const dpUnitPrice = toMoney(item.dpPrice ?? item.dp_unit_price ?? item.distributorPrice);
  const chargedUnitPrice = toMoney(item.salePrice ?? item.chargedPrice ?? item.cpPrice ?? dpUnitPrice);

  return {
    productId: item.productId,
    productName: item.productName,
    batchNo: item.batchNo ?? null,
    expiryDate: item.expiryDate ?? null,
    quantity,
    dpUnitPrice,
    chargedUnitPrice,
    dpTotal: toMoney(dpUnitPrice * quantity),
    chargedTotal: toMoney(chargedUnitPrice * quantity),
    branchProfit: toMoney((chargedUnitPrice - dpUnitPrice) * quantity),
    bvTotal: toMoney((Number(item.bv || 0)) * quantity),
  };
});

const dpTotal = toMoney(saleLines.reduce((sum, line) => sum + line.dpTotal, 0));
const chargedTotal = toMoney(saleLines.reduce((sum, line) => sum + line.chargedTotal, 0));
const branchProfitTotal = toMoney(saleLines.reduce((sum, line) => sum + line.branchProfit, 0));
```

## SQL/Data Migration Guidance

If using Postgres/Supabase, add missing columns if they do not exist:

```sql
alter table stock_movements
  add column if not exists dp_unit_price numeric,
  add column if not exists dp_value numeric,
  add column if not exists charged_unit_price numeric,
  add column if not exists charged_value numeric,
  add column if not exists branch_profit_value numeric,
  add column if not exists receipt_no text,
  add column if not exists business_date date,
  add column if not exists timezone text default 'Africa/Windhoek';

alter table sales
  add column if not exists dp_total numeric,
  add column if not exists charged_total numeric,
  add column if not exists branch_profit_total numeric,
  add column if not exists business_date date,
  add column if not exists timezone text default 'Africa/Windhoek';

alter table sale_lines
  add column if not exists dp_unit_price numeric,
  add column if not exists dp_total numeric,
  add column if not exists charged_unit_price numeric,
  add column if not exists charged_total numeric,
  add column if not exists branch_profit numeric;
```

Backfill blank sale movement `dp_value` from sale lines where possible:

```sql
update stock_movements sm
set
  dp_unit_price = coalesce(sm.dp_unit_price, sl.dp_unit_price),
  dp_value = coalesce(sm.dp_value, sl.dp_total, sl.dp_unit_price * abs(sm.quantity)),
  charged_unit_price = coalesce(sm.charged_unit_price, sl.charged_unit_price),
  charged_value = coalesce(sm.charged_value, sl.charged_total),
  branch_profit_value = coalesce(sm.branch_profit_value, sl.branch_profit),
  receipt_no = coalesce(sm.receipt_no, s.receipt_no),
  business_date = coalesce(sm.business_date, s.business_date, (sm.created_at at time zone 'Africa/Windhoek')::date),
  timezone = coalesce(sm.timezone, 'Africa/Windhoek')
from sales s
join sale_lines sl on sl.sale_id = s.id
where sm.reference = s.reference
  and sm.product_id = sl.product_id
  and sm.movement_type in ('non_member_sale', 'member_sale', 'prescription_dispense', 'campaign_sale')
  and (sm.dp_value is null or sm.dp_value = 0);
```

If exact keys differ, match by `receipt_no`, `sale_id`, `reference`, `product_id`, `product_name`, and timestamp proximity.

Backfill MIS audit logs for completed sales that are missing audit entries:

```sql
insert into audit_logs (
  event_type,
  branch,
  reference,
  receipt_no,
  user_id,
  username,
  created_at,
  business_date,
  timezone,
  details
)
select
  'sale_created',
  s.branch,
  s.reference,
  s.receipt_no,
  s.staff_user_id,
  s.staff_username,
  s.created_at,
  coalesce(s.business_date, (s.created_at at time zone 'Africa/Windhoek')::date),
  'Africa/Windhoek',
  jsonb_build_object(
    'customerType', s.customer_type,
    'paymentMethod', s.payment_method,
    'customerName', s.customer_name,
    'dpTotal', s.dp_total,
    'chargedTotal', s.charged_total,
    'branchProfitTotal', s.branch_profit_total,
    'source', 'backfill_missing_sale_created_audit'
  )
from sales s
where not exists (
  select 1
  from audit_logs a
  where a.event_type = 'sale_created'
    and (
      a.reference = s.reference
      or a.receipt_no = s.receipt_no
    )
);
```

## Fix Stock Portal Queries

- `salesRevenue`
- `branchRevenue`
- `stockRevenue`
- `movementValue`
- `dpOut`
- `DP VALUE`

For stock-linked sales reporting, replace charged/CP calculations with DP calculations:

```ts
// Wrong for stock DP reporting:
sum += movement.charged_value ?? movement.customer_paid ?? movement.total;

// Correct:
sum += movement.dp_value ?? (movement.dp_unit_price * Math.abs(movement.quantity));
```

Keep charged values only where the UI clearly says `Customer charged revenue`, `CP receipts`, or `cash collected`.

## Fix Open Requests Count

Find the logic behind `OPEN REQUESTS`.

It must only count stock request records:

```ts
const OPEN_STOCK_REQUEST_STATUSES = new Set([
  "pending",
  "approved",
  "in_transit",
  "partially_received",
]);

const openRequests = stockRequests.filter((request) =>
  request.branch === selectedBranch &&
  OPEN_STOCK_REQUEST_STATUSES.has(String(request.status || "").toLowerCase())
).length;
```

It must not count:

- sale movements
- reorder recommendations
- low-stock alerts
- stock movement ledger rows
- branch inventory rows

Add a regression test:

```ts
it("does not increment open stock requests when completing a walk-in sale", async () => {
  const before = await getOpenStockRequestCount("Tsumeb");
  await completeWalkInSale({
    branch: "Tsumeb",

    quantity: 1,
    paymentMethod: "cash",
  });
  const after = await getOpenStockRequestCount("Tsumeb");
  expect(after).toBe(before);
});
```
